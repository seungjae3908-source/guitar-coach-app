function scoreSession({
  attacks,
  beats,
  noiseFloor,
  clippingCount,
  manualMistakes,
  targetSeconds,
  practicedSeconds,
  metronomeSound,
}: {
  attacks: Attack[];
  beats: BeatEvent[];
  noiseFloor: number;
  clippingCount: number;
  manualMistakes: number;
  targetSeconds: number;
  practicedSeconds: number;
  metronomeSound: boolean;
}) {
  const matched = attacks.filter((attack) => attack.errorMs !== undefined);
  const errors = matched.map((attack) => Math.abs(attack.errorMs ?? 0));
  const signedErrors = matched.map((attack) => attack.errorMs ?? 0);
  const expectedBeats = Math.max(1, beats.length);
  const coverage = clamp((matched.length / expectedBeats) * 100);
  const meanError = average(errors);
  const rhythmAccuracy = clamp(100 - meanError / 2.4) * 0.7 + coverage * 0.3;
  const timingConsistency = clamp(100 - standardDeviation(signedErrors) / 2.1);
  const levelValues = attacks.map((attack) => attack.levelDb).filter(Number.isFinite);
  const dynamicStability = levelValues.length >= 2 ? clamp(100 - standardDeviation(levelValues) * 11) : 35;
  const noiseControl = clamp(((-noiseFloor - 24) / 36) * 100 - clippingCount * 3);
  const completion = clamp((practicedSeconds / Math.max(1, targetSeconds)) * 100);
  const mistakePenalty = Math.min(24, manualMistakes * 4);
  const confidence = clamp(
    35 + Math.min(35, attacks.length * 2.2) + Math.min(25, beats.length * 0.8) - (metronomeSound ? 8 : 0),
    20,
    96,
  );
  const total = clamp(
    rhythmAccuracy * 0.34 +
      timingConsistency * 0.2 +
      dynamicStability * 0.2 +
      noiseControl * 0.11 +
      coverage * 0.1 +
      completion * 0.05 -
      mistakePenalty,
  );

  return {
    total: Math.round(total),
    rhythmAccuracy: Math.round(rhythmAccuracy),
    timingConsistency: Math.round(timingConsistency),
    dynamicStability: Math.round(dynamicStability),
    noiseControl: Math.round(noiseControl),
    detectedAttacks: attacks.length,
    matchedBeats: matched.length,
    expectedBeats,
    averageOffsetMs: Math.round(average(signedErrors)),
    confidence: Math.round(confidence),
    coverage: Math.round(coverage),
  };
}

function chooseFeedback(mode: FocusMode, scores: ReturnType<typeof scoreSession>) {
  const config = MODE_CONFIG[mode];
  const candidates = [
    { key: 'rhythm', value: scores.rhythmAccuracy },
    { key: 'timing', value: scores.timingConsistency },
    { key: 'dynamics', value: scores.dynamicStability },
    { key: 'noise', value: scores.noiseControl },
    { key: 'continuity', value: scores.coverage },
  ].sort((a, b) => a.value - b.value);

  if (scores.total >= 88 && scores.coverage >= 80) return config.feedback.strong;
  const weakest = candidates[0]?.key;
  if (weakest === 'dynamics') return config.feedback.dynamics;
  if (weakest === 'noise') return config.feedback.noise;
  if (weakest === 'continuity') return config.feedback.continuity;
  return config.feedback.rhythm;
}

  );
}
